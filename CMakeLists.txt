cmake_minimum_required(VERSION 3.16)
project(PlacementLens LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PLACEMENTLENS_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(PLACEMENTLENS_TSAN "Enable thread sanitizer (separate from ASan/UBSan)" OFF)
option(PLACEMENTLENS_OPENCL "Enable the optional real OpenCL GPU backend" OFF)
if(PLACEMENTLENS_SANITIZERS AND PLACEMENTLENS_TSAN)
  message(FATAL_ERROR "ThreadSanitizer requires a separate build from ASan/UBSan")
endif()
find_package(Threads REQUIRED)
add_library(placementlens_core src/core.cpp src/report.cpp src/async.cpp src/queue.cpp src/opencl_device.cpp src/kernels.cpp src/cnn.cpp src/search.cpp)
target_include_directories(placementlens_core PUBLIC include)
target_link_libraries(placementlens_core PUBLIC Threads::Threads)
if(PLACEMENTLENS_OPENCL)
  find_package(OpenCL REQUIRED)
  target_compile_definitions(placementlens_core PRIVATE PL_HAVE_OPENCL=1)
  target_link_libraries(placementlens_core PUBLIC OpenCL::OpenCL)
endif()
if(MSVC)
  target_compile_options(placementlens_core PUBLIC /W4)
  if(PLACEMENTLENS_SANITIZERS OR PLACEMENTLENS_TSAN)
    message(FATAL_ERROR "These sanitizer combinations are configured only for Clang/GCC")
  endif()
else()
  target_compile_options(placementlens_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(PLACEMENTLENS_SANITIZERS)
    target_compile_options(placementlens_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(placementlens_core PUBLIC -fsanitize=address,undefined)
  elseif(PLACEMENTLENS_TSAN)
    target_compile_options(placementlens_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(placementlens_core PUBLIC -fsanitize=thread)
  endif()
endif()
add_executable(placementlens src/main.cpp)
target_link_libraries(placementlens PRIVATE placementlens_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(placementlens_tests tests/core_tests.cpp)
  target_link_libraries(placementlens_tests PRIVATE placementlens_core)
  add_test(NAME core_contracts COMMAND placementlens_tests)
  add_executable(async_tests tests/async_tests.cpp)
  target_link_libraries(async_tests PRIVATE placementlens_core)
  add_test(NAME async_contracts COMMAND async_tests)
  set_tests_properties(async_contracts PROPERTIES TIMEOUT 30)
  add_executable(queue_tests tests/queue_tests.cpp)
  target_link_libraries(queue_tests PRIVATE placementlens_core)
  add_test(NAME queue_contracts COMMAND queue_tests "${CMAKE_SOURCE_DIR}")
  set_tests_properties(queue_contracts PROPERTIES TIMEOUT 60)
  add_executable(opencl_tests tests/opencl_tests.cpp)
  target_link_libraries(opencl_tests PRIVATE placementlens_core)
  add_test(NAME opencl_contracts COMMAND opencl_tests)
  set_tests_properties(opencl_contracts PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 90)
  add_executable(device_dispatch_tests tests/device_dispatch_tests.cpp)
  target_link_libraries(device_dispatch_tests PRIVATE placementlens_core)
  add_test(NAME device_dispatch_contracts COMMAND device_dispatch_tests)
  foreach(suite kernel memory residency runtime_review cnn search)
    add_executable(${suite}_tests tests/${suite}_tests.cpp)
    target_link_libraries(${suite}_tests PRIVATE placementlens_core)
    add_test(NAME ${suite}_contracts COMMAND ${suite}_tests)
  endforeach()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME search_cli_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/search_cli_tests.py" --binary "$<TARGET_FILE:placementlens>")
    add_test(NAME bounded_study_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/bounded_study_tests.py" "$<TARGET_FILE:placementlens>")
    add_test(NAME cli_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/cli_tests.py" "$<TARGET_FILE:placementlens>" "${CMAKE_SOURCE_DIR}")
    add_test(NAME placement_matrix COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/check_matrix.py" --binary "$<TARGET_FILE:placementlens>" --out "${CMAKE_BINARY_DIR}/verification")
    add_test(NAME differential_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/differential_tests.py" "$<TARGET_FILE:placementlens>" "${CMAKE_BINARY_DIR}/differential")
    add_test(NAME measured_study_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/study_tests.py" "$<TARGET_FILE:placementlens>")
    add_test(NAME production_reference_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/production_tests.py" "$<TARGET_FILE:placementlens>" "${CMAKE_SOURCE_DIR}")
    set_tests_properties(production_reference_contracts PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
    add_test(NAME workbench_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/workbench_tests.py" "$<TARGET_FILE:placementlens>")
    set_tests_properties(workbench_contracts PROPERTIES TIMEOUT 180)
    add_test(NAME board_probe_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/board_probe_tests.py")
    add_test(NAME board_handoff_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/board_handoff_tests.py")
    add_test(NAME target_profile_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/target_profile_tests.py")
    add_test(NAME ti_adapter_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/ti_adapter_tests.py")
    set_tests_properties(ti_adapter_contracts PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    foreach(suite onnx_frontend onnx_review model_explorer developer_cli)
      add_test(NAME ${suite}_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/${suite}_tests.py" --binary "$<TARGET_FILE:placementlens>")
      set_tests_properties(${suite}_contracts PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
    endforeach()
    add_test(NAME trained_example_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/trained_example_tests.py" --binary "$<TARGET_FILE:placementlens>")
    set_tests_properties(trained_example_contracts PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300)
    add_test(NAME architecture_model_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/architecture_model_tests.py" --binary "$<TARGET_FILE:placementlens>")
    set_tests_properties(architecture_model_contracts PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300)
  else()
    message(WARNING "Python3 unavailable: CLI, placement matrix and differential checks will not run")
  endif()
endif()
